def load_docs(path: str, fallback_url: str) -> List[str]:
    if not os.path.exists(path):
        urllib.request.urlretrieve(fallback_url, path)
    with open(path, "r", encoding="utf-8") as f:
        docs = [line.strip() for line in f if line.strip()]
    random.shuffle(docs)
    return docs


class CharTokenizer:
    def __init__(self, docs: Sequence[str]):
        self.uchars = sorted(set("".join(docs)))
        self.BOS = len(self.uchars)
        self.vocab_size = len(self.uchars) + 1
        self.stoi = {c: i for i, c in enumerate(self.uchars)}

    def encode(self, s: str) -> List[int]:
        return [self.stoi[c] for c in s]

    def encode_with_bos(self, doc: str) -> List[int]:
        return [self.BOS] + self.encode(doc) + [self.BOS]

    def state_dict(self) -> Dict[str, Any]:
        return {"uchars": self.uchars}

    @classmethod
    def load_state_dict(cls, state: Dict[str, Any]) -> "CharTokenizer":
        obj = cls.__new__(cls)
        obj.uchars = list(state["uchars"])
        obj.BOS = len(obj.uchars)
        obj.vocab_size = len(obj.uchars) + 1
        obj.stoi = {c: i for i, c in enumerate(obj.uchars)}
        return obj


class NameSampler:
    def __init__(self, encoded_docs: Sequence[Sequence[int]], block_size: int, random_offset: bool = False):
        self.encoded_docs = list(encoded_docs)
        self.block_size = int(block_size)
        self.random_offset = bool(random_offset)
        self._idx = 0

    def next_one(self) -> Tuple[List[int], List[int]]:
        tokens = self.encoded_docs[self._idx]
        self._idx = (self._idx + 1) % len(self.encoded_docs)
        if len(tokens) < 2:
            return [tokens[0]], [tokens[0]]
        start = random.randint(0, len(tokens) - 2) if self.random_offset else 0
        max_T = len(tokens) - 1 - start
        T = min(self.block_size, max_T)
        x = list(tokens[start : start + T])
        y = list(tokens[start + 1 : start + T + 1])
        return x, y

    def next_batch(self, batch_size: int) -> Tuple[List[List[int]], List[List[int]]]:
        xs, ys = [], []
        for _ in range(batch_size):
            x, y = self.next_one()
            xs.append(x); ys.append(y)
        T = min(len(x) for x in xs)
        xs = [x[:T] for x in xs]
        ys = [y[:T] for y in ys]
        return xs, ys

def build_backend_list():
    class Value:
        __slots__ = ("data", "grad", "_children", "_local_grads")
        TRACK_GRAD = True

        def __init__(self, data: float, children=(), local_grads=()):
            self.data = float(data)
            self.grad = 0.0
            if Value.TRACK_GRAD and children:
                self._children = children
                self._local_grads = local_grads
            else:
                self._children = ()
                self._local_grads = ()

        def __add__(self, other):
            other = other if isinstance(other, Value) else Value(other)
            if not Value.TRACK_GRAD: return Value(self.data + other.data)
            return Value(self.data + other.data, (self, other), (1, 1))

        def __mul__(self, other):
            other = other if isinstance(other, Value) else Value(other)
            if not Value.TRACK_GRAD: return Value(self.data * other.data)
            return Value(self.data * other.data, (self, other), (other.data, self.data))

        def __pow__(self, other: float):
            if not Value.TRACK_GRAD: return Value(self.data**other)
            return Value(self.data**other, (self,), (other * self.data ** (other - 1),))

        def __neg__(self): return self * -1
        def __sub__(self, other): return self + (-other)
        def __radd__(self, other): return self + other
        def __rmul__(self, other): return self * other
        def __truediv__(self, other): return self * (other ** -1)

        def log(self):
            if not Value.TRACK_GRAD: return Value(math.log(self.data))
            return Value(math.log(self.data), (self,), (1 / self.data,))

        def exp(self):
            e = math.exp(self.data)
            if not Value.TRACK_GRAD: return Value(e)
            return Value(e, (self,), (e,))

        def relu(self):
            out = max(0.0, self.data)
            if not Value.TRACK_GRAD: return Value(out)
            return Value(out, (self,), (float(self.data > 0),))

        def backward(self):
            if not Value.TRACK_GRAD:
                raise RuntimeError("backward() with TRACK_GRAD=False")
            stack, flags, topo, seen = [self], [0], [], set()
            while stack:
                v, f = stack.pop(), flags.pop()
                vid = id(v)
                if f:
                    topo.append(v); continue
                if vid in seen: continue
                seen.add(vid)
                stack.append(v); flags.append(1)
                for c in v._children:
                    stack.append(c); flags.append(0)
            self.grad = 1.0
            for v in reversed(topo):
                gv = v.grad
                if gv == 0.0: continue
                for c, lg in zip(v._children, v._local_grads):
                    c.grad += lg * gv

    class NoGrad:
        def __enter__(self):
            self.prev = Value.TRACK_GRAD
            Value.TRACK_GRAD = False
        def __exit__(self, exc_type, exc, tb):
            Value.TRACK_GRAD = self.prev

    class Parameter(Value):
        pass

    def _iter_parameters(obj) -> Iterator[Parameter]:
        if isinstance(obj, Parameter):
            yield obj
        elif isinstance(obj, Tensor):
            yield from obj.parameters()
        elif isinstance(obj, list):
            for it in obj: yield from _iter_parameters(it)
        elif isinstance(obj, tuple):
            for it in obj: yield from _iter_parameters(it)

    class Tensor:
        def __init__(self, data, shape: Tuple[int, ...]):
            self.data = data
            self._shape = tuple(int(x) for x in shape)

        @property
        def shape(self): return self._shape

        def parameters(self) -> Iterator[Parameter]:
            yield from _iter_parameters(self.data)

        @staticmethod
        def randn(shape: Tuple[int, ...], std: float = 0.08, as_param: bool = False) -> "Tensor":
            def make(level: int):
                if level == len(shape) - 1:
                    row = []
                    for _ in range(shape[level]):
                        v = random.gauss(0.0, std)
                        row.append(Parameter(v) if as_param else Value(v))
                    return row
                return [make(level + 1) for _ in range(shape[level])]
            return Tensor(make(0), shape)

        # ---- 1D (for KV sampling)
        def row2d(self, i: int) -> "Tensor":
            assert len(self._shape) == 2
            return Tensor(self.data[int(i)], (self._shape[1],))

        @staticmethod
        def from_1d(xs) -> "Tensor":
            xs = list(xs)
            return Tensor(xs, (len(xs),))

        @staticmethod
        def matvec(w: "Tensor", x: "Tensor") -> "Tensor":
            O, I = w.shape
            out=[]
            for o in range(O):
                s=Value(0.0)
                for i in range(I):
                    s = s + w.data[o][i] * x.data[i]
                out.append(s)
            return Tensor(out, (O,))

        def softmax_1d(self) -> "Tensor":
            m = max(v.data for v in self.data)
            exps = [(v - m).exp() for v in self.data]
            tot = Value(0.0)
            for e in exps: tot = tot + e
            return Tensor([e / tot for e in exps], self.shape)

        @staticmethod
        def rmsnorm_1d(x: "Tensor", eps: float) -> "Tensor":
            C = x.shape[0]
            ms = Value(0.0)
            for j in range(C):
                ms = ms + x.data[j]*x.data[j]
            ms = ms / C
            scale = (ms + eps) ** -0.5
            return Tensor([x.data[j]*scale for j in range(C)], x.shape)

        @staticmethod
        def relu_1d(x: "Tensor") -> "Tensor":
            return Tensor([v.relu() for v in x.data], x.shape)

        @staticmethod
        def split_heads_1d(x: "Tensor", n_head: int) -> List["Tensor"]:
            C = x.shape[0]; Hd = C // n_head
            return [Tensor(x.data[h*Hd:(h+1)*Hd], (Hd,)) for h in range(n_head)]

        @staticmethod
        def cat_heads_1d(heads: List["Tensor"]) -> "Tensor":
            data=[]
            for h in heads: data.extend(h.data)
            return Tensor(data, (len(data),))

        # ---- batch ops (training)
        @staticmethod
        def gather_rows_batch(weight_vc: "Tensor", idx_bt: List[List[int]]) -> "Tensor":
            B = len(idx_bt); T = len(idx_bt[0])
            out=[]
            for b in range(B):
                rows=[]
                for t in range(T):
                    rows.append(list(weight_vc.data[int(idx_bt[b][t])]))
                out.append(rows)
            return Tensor(out, (B, T, weight_vc.shape[1]))

        @staticmethod
        def add_btc(a: "Tensor", b: "Tensor") -> "Tensor":
            B, T, C = a.shape
            assert b.shape == (B, T, C)
            out=[]
            for bb in range(B):
                rows=[]
                for tt in range(T):
                    rows.append([a.data[bb][tt][j] + b.data[bb][tt][j] for j in range(C)])
                out.append(rows)
            return Tensor(out, (B, T, C))

        @staticmethod
        def rmsnorm_btc(x: "Tensor", eps: float) -> "Tensor":
            B, T, C = x.shape
            out=[]
            for bb in range(B):
                rows=[]
                for tt in range(T):
                    row = x.data[bb][tt]
                    ms = Value(0.0)
                    for j in range(C):
                        ms = ms + row[j]*row[j]
                    ms = ms / C
                    scale = (ms + eps) ** -0.5
                    rows.append([row[j]*scale for j in range(C)])
                out.append(rows)
            return Tensor(out, (B, T, C))

        @staticmethod
        def mul_gamma_btc(x: "Tensor", gamma_c: "Tensor") -> "Tensor":
            B, T, C = x.shape
            assert gamma_c.shape == (C,)
            out=[]
            for bb in range(B):
                rows=[]
                for tt in range(T):
                    rows.append([x.data[bb][tt][j] * gamma_c.data[j] for j in range(C)])
                out.append(rows)
            return Tensor(out, (B, T, C))

        @staticmethod
        def linear_btc(x_bti: "Tensor", w_oi: "Tensor") -> "Tensor":
            B, T, I = x_bti.shape
            O, I2 = w_oi.shape
            assert I == I2
            out=[]
            for bb in range(B):
                rows=[]
                for tt in range(T):
                    xrow = x_bti.data[bb][tt]
                    yrow=[]
                    for o in range(O):
                        s=Value(0.0)
                        wrow = w_oi.data[o]
                        for i in range(I):
                            s = s + wrow[i]*xrow[i]
                        yrow.append(s)
                    rows.append(yrow)
                out.append(rows)
            return Tensor(out, (B, T, O))

        @staticmethod
        def relu_btc(x: "Tensor") -> "Tensor":
            B, T, C = x.shape
            out=[]
            for bb in range(B):
                rows=[]
                for tt in range(T):
                    rows.append([x.data[bb][tt][j].relu() for j in range(C)])
                out.append(rows)
            return Tensor(out, (B, T, C))

        @staticmethod
        def reshape_bthd(x_btc: "Tensor", n_head: int) -> "Tensor":
            B, T, C = x_btc.shape
            Hd = C // n_head
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(n_head):
                    rows=[]
                    for tt in range(T):
                        rows.append(x_btc.data[bb][tt][h*Hd:(h+1)*Hd])
                    hh.append(rows)
                out.append(hh)
            return Tensor(out, (B, n_head, T, Hd))

        @staticmethod
        def transpose_last2(x_bhtd: "Tensor") -> "Tensor":
            B, H, T, Hd = x_bhtd.shape
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(H):
                    mat=[]
                    for d in range(Hd):
                        mat.append([x_bhtd.data[bb][h][t][d] for t in range(T)])
                    hh.append(mat)
                out.append(hh)
            return Tensor(out, (B, H, Hd, T))

        @staticmethod
        def matmul_bhtt(a_bhtd: "Tensor", b_bhdt: "Tensor") -> "Tensor":
            B, H, T, Hd = a_bhtd.shape
            B2, H2, Hd2, T2 = b_bhdt.shape
            assert (B, H, Hd, T) == (B2, H2, Hd2, T2)
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(H):
                    mat=[]
                    for i in range(T):
                        row=[]
                        for j in range(T):
                            s=Value(0.0)
                            for d in range(Hd):
                                s = s + a_bhtd.data[bb][h][i][d] * b_bhdt.data[bb][h][d][j]
                            row.append(s)
                        mat.append(row)
                    hh.append(mat)
                out.append(hh)
            return Tensor(out, (B, H, T, T))

        @staticmethod
        def scale_bhtt(x: "Tensor", s: float) -> "Tensor":
            B, H, T, T2 = x.shape
            assert T == T2
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(H):
                    mat=[]
                    for i in range(T):
                        mat.append([x.data[bb][h][i][j]*s for j in range(T)])
                    hh.append(mat)
                out.append(hh)
            return Tensor(out, (B, H, T, T))

        @staticmethod
        def causal_mask_bhtt(x: "Tensor", mask_value: float = -1e9) -> "Tensor":
            B, H, T, T2 = x.shape
            assert T == T2
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(H):
                    mat=[]
                    for i in range(T):
                        row=[]
                        for j in range(T):
                            row.append(Value(mask_value) if j > i else x.data[bb][h][i][j])
                        mat.append(row)
                    hh.append(mat)
                out.append(hh)
            return Tensor(out, (B, H, T, T))

        @staticmethod
        def softmax_last_bhtt(x: "Tensor") -> "Tensor":
            B, H, T, T2 = x.shape
            assert T == T2
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(H):
                    mat=[]
                    for i in range(T):
                        row = x.data[bb][h][i]
                        maxv = max(v.data for v in row)
                        exps = [(v - maxv).exp() for v in row]
                        tot = Value(0.0)
                        for e in exps: tot = tot + e
                        mat.append([e/tot for e in exps])
                    hh.append(mat)
                out.append(hh)
            return Tensor(out, (B, H, T, T))

        @staticmethod
        def matmul_bhtd(w_bhtt: "Tensor", v_bhtd: "Tensor") -> "Tensor":
            B, H, T, T2 = w_bhtt.shape
            B2, H2, T3, Hd = v_bhtd.shape
            assert (B, H, T) == (B2, H2, T3) and T2 == T
            out=[]
            for bb in range(B):
                hh=[]
                for h in range(H):
                    rows=[]
                    for i in range(T):
                        outrow=[]
                        for d in range(Hd):
                            s=Value(0.0)
                            for j in range(T):
                                s = s + w_bhtt.data[bb][h][i][j] * v_bhtd.data[bb][h][j][d]
                            outrow.append(s)
                        rows.append(outrow)
                    hh.append(rows)
                out.append(hh)
            return Tensor(out, (B, H, T, Hd))

        @staticmethod
        def merge_heads_btc(x_bhtd: "Tensor") -> "Tensor":
            B, H, T, Hd = x_bhtd.shape
            out=[]
            for bb in range(B):
                rows=[]
                for tt in range(T):
                    row=[]
                    for h in range(H):
                        row.extend(x_bhtd.data[bb][h][tt])
                    rows.append(row)
                out.append(rows)
            return Tensor(out, (B, T, H*Hd))

    def clip_grad_norm(params: List[Parameter], max_norm: float) -> float:
        if max_norm <= 0: return 0.0
        sq = 0.0
        for p in params:
            sq += p.grad * p.grad
        norm = sq ** 0.5
        if norm > max_norm and norm > 0:
            scale = max_norm / norm
            for p in params:
                p.grad *= scale
        return norm

    def pack_floats(x):
        if isinstance(x, Parameter): return float(x.data)
        if isinstance(x, Value): return float(x.data)
        if isinstance(x, Tensor): return {"shape": x.shape, "data": pack_floats(x.data)}
        if isinstance(x, list): return [pack_floats(v) for v in x]
        return x

    def unpack_tensor_like(t: Tensor, payload):
        def assign(dst, src):
            if isinstance(dst, list) and isinstance(src, list):
                for d, s in zip(dst, src): assign(d, s)
            elif isinstance(dst, Parameter):
                dst.data = float(src)
            elif isinstance(dst, Value):
                dst.data = float(src)
        assign(t.data, payload["data"])

    return {
        "kind": "list",
        "Value": Value,
        "NoGrad": NoGrad,
        "Parameter": Parameter,
        "Tensor": Tensor,
        "clip_grad_norm": clip_grad_norm,
        "pack_floats": pack_floats,
        "unpack_tensor_like": unpack_tensor_like,
    }

def build_backend_torch(device: str):
    import torch
    import torch.nn.functional as F
    
    torch.manual_seed(42)
    if device.startswith("cuda"): torch.cuda.manual_seed_all(42)

    class NoGrad:
        def __enter__(self):
            self.ctx = torch.no_grad()
            self.ctx.__enter__()
        def __exit__(self, exc_type, exc, tb):
            self.ctx.__exit__(exc_type, exc, tb)

    class Tensor:
        def __init__(self, data: torch.Tensor):
            self.data = data

        @property
        def shape(self): return tuple(self.data.shape)

        @staticmethod
        def randn(shape: Tuple[int, ...], std: float = 0.08, as_param: bool = False) -> "Tensor":
            t = (torch.randn(*shape, device=device) * std)
            if as_param: return Tensor(torch.nn.Parameter(t))
            return Tensor(t)

        def row(self, i: int) -> "Tensor":
            return Tensor(self.data[i])

        @staticmethod
        def matvec(w: "Tensor", x: "Tensor") -> "Tensor":
            return Tensor(w.data @ x.data)

        def softmax_1d(self) -> "Tensor":
            return Tensor(F.softmax(self.data, dim=-1))

    def unpack_param_like(param, payload):
        import torch
        t = torch.tensor(payload, device=device, dtype=param.data.dtype)
        if tuple(param.data.shape) != tuple(t.shape):
            t = t.view_as(param.data)
        with torch.no_grad():
            param.data.copy_(t)

    return {
        "kind": "torch",
        "torch": __import__("torch"),
        "F": __import__("torch.nn.functional", fromlist=["*"]),
        "device": device,
        "NoGrad": NoGrad,
        "Tensor": Tensor,
        "unpack_param_like": unpack_param_like,
    }

# Adam unified (named) helper
# ============================================================
def _param_numel_and_flatten_kind(B, p):
    if B["kind"] == "torch":
        torch = B["torch"]
        n = int(p.numel())
        def flat(): return p.detach().float().reshape(-1).cpu().tolist()
        def assign(xs):
            t = torch.tensor(xs, device=B["device"], dtype=p.data.dtype).view_as(p.data)
            with torch.no_grad():
                p.data.copy_(t)
        return n, "torch", flat, assign

    def iter_leaf(x):
        if hasattr(x, "grad") and hasattr(x, "data"):
            yield x
        elif isinstance(x, list):
            for it in x: yield from iter_leaf(it)
        else:
            raise TypeError(type(x))
    leaves = list(iter_leaf(p))
    n = len(leaves)
    def flat(): return [float(pp.data) for pp in leaves]
    def assign(xs):
        assert len(xs) == n
        for pp, v in zip(leaves, xs):
            pp.data = float(v)
    return n, "list", flat, assign


class AdamUnifiedNamedTorch:
    def __init__(self, B, named_params, lr, beta1, beta2, eps):
        self.B=B; self.torch=B["torch"]
        self.lr=float(lr); self.beta1=float(beta1); self.beta2=float(beta2); self.eps=float(eps)
        self.step_idx=0
        self.named_params=sorted(list(named_params), key=lambda x:x[0])
        self.m={}; self.v={}
        for name,p in self.named_params:
            n,_,_,_=_param_numel_and_flatten_kind(B,p)
            self.m[name]=[0.0]*n; self.v[name]=[0.0]*n

    def unified_state(self):
        return {"kind":"adam_unified_named_v1","step":self.step_idx,"m":self.m,"v":self.v}

    def load_unified_state(self,u):
        if not u: return
        if u.get("kind")!="adam_unified_named_v1":
            raise ValueError(f"Unsupported opt state kind: {u.get('kind')}")
        self.step_idx=int(u["step"])
        um,uv=u.get("m",{}),u.get("v",{})
        for name,p in self.named_params:
            if name in um and name in uv:
                n,_,_,_=_param_numel_and_flatten_kind(self.B,p)
                if len(um[name])!=n or len(uv[name])!=n:
                    raise ValueError(f"Adam state mismatch at {name}: expected {n}")
                self.m[name]=list(um[name]); self.v[name]=list(uv[name])

    def zero_grad(self):
        for _,p in self.named_params:
            p.grad=None

    def step(self, lr_scale=1.0):
        torch=self.torch
        self.step_idx+=1
        lr_t=self.lr*lr_scale
        with torch.no_grad():
            for name,p in self.named_params:
                if p.grad is None: continue
                g=p.grad.detach().float().reshape(-1).cpu().tolist()
                m=self.m[name]; v=self.v[name]
                n=len(g)
                for i in range(n):
                    gi=g[i]
                    mi=m[i]=self.beta1*m[i]+(1-self.beta1)*gi
                    vi=v[i]=self.beta2*v[i]+(1-self.beta2)*(gi*gi)
                    m_hat=mi/(1-self.beta1**self.step_idx)
                    v_hat=vi/(1-self.beta2**self.step_idx)
                    g[i]=m_hat/(math.sqrt(v_hat)+self.eps)
                upd=torch.tensor(g,device=self.B["device"],dtype=p.data.dtype).view_as(p.data)
                p.data.add_(upd,alpha=-lr_t)
                p.grad=None


class AdamUnifiedNamedList:
    def __init__(self, B, named_params, lr, beta1, beta2, eps):
        self.B=B
        self.lr=float(lr); self.beta1=float(beta1); self.beta2=float(beta2); self.eps=float(eps)
        self.step_idx=0
        self.named_params=sorted(list(named_params), key=lambda x:x[0])
        self.m={}; self.v={}
        for name,p in self.named_params:
            n,_,_,_=_param_numel_and_flatten_kind(B,p)
            self.m[name]=[0.0]*n; self.v[name]=[0.0]*n

    def unified_state(self):
        return {"kind":"adam_unified_named_v1","step":self.step_idx,"m":self.m,"v":self.v}

    def load_unified_state(self,u):
        if not u: return
        if u.get("kind")!="adam_unified_named_v1":
            raise ValueError(f"Unsupported opt state kind: {u.get('kind')}")
        self.step_idx=int(u["step"])
        um,uv=u.get("m",{}),u.get("v",{})
        for name,p in self.named_params:
            if name in um and name in uv:
                n,_,_,_=_param_numel_and_flatten_kind(self.B,p)
                if len(um[name])!=n or len(uv[name])!=n:
                    raise ValueError(f"Adam state mismatch at {name}: expected {n}")
                self.m[name]=list(um[name]); self.v[name]=list(uv[name])

    def step(self, lr_scale=1.0):
        self.step_idx+=1
        lr_t=self.lr*lr_scale

        def iter_leaf(x):
            if hasattr(x,"grad") and hasattr(x,"data"):
                yield x
            elif isinstance(x,list):
                for it in x: yield from iter_leaf(it)
            else:
                raise TypeError(type(x))

        for name,root in self.named_params:
            leaves=list(iter_leaf(root))
            m=self.m[name]; v=self.v[name]
            for i,leaf in enumerate(leaves):
                g=leaf.grad
                m[i]=self.beta1*m[i]+(1-self.beta1)*g
                v[i]=self.beta2*v[i]+(1-self.beta2)*(g**2)
                m_hat=m[i]/(1-self.beta1**self.step_idx)
                v_hat=v[i]/(1-self.beta2**self.step_idx)
                leaf.data-=lr_t*m_hat/(math.sqrt(v_hat)+self.eps)
                leaf.grad=0.0

class KVCache:
    def __init__(self, n_layer: int, n_head: int):
        self.n_layer=int(n_layer); self.n_head=int(n_head)
        self.keys=[[[] for _ in range(n_head)] for _ in range(n_layer)]
        self.values=[[[] for _ in range(n_head)] for _ in range(n_layer)]

    def append(self, layer_idx: int, k_heads: List[Any], v_heads: List[Any]):
        for h in range(self.n_head):
            self.keys[layer_idx][h].append(k_heads[h])
            self.values[layer_idx][h].append(v_heads[h])


class TorchKVCachePrealloc:
    # Prealloc per layer:
    # k[layer], v[layer]: [B,H,block_size,Hd]
    # current length t
    def __init__(self, torch, device: str, n_layer: int, n_head: int, head_dim: int, batch_size: int, block_size: int, dtype):
        self.torch = torch
        self.device = device
        self.n_layer = int(n_layer)
        self.n_head = int(n_head)
        self.head_dim = int(head_dim)
        self.batch_size = int(batch_size)
        self.block_size = int(block_size)
        self.dtype = dtype
        self.t = 0
        self.k = [
            torch.empty((batch_size, n_head, block_size, head_dim), device=device, dtype=dtype)
            for _ in range(n_layer)
        ]
        self.v = [
            torch.empty((batch_size, n_head, block_size, head_dim), device=device, dtype=dtype)
            for _ in range(n_layer)
        ]

    def append(self, layer_idx: int, k_new, v_new):
        # k_new, v_new: [B,H,1,Hd]
        t = self.t
        self.k[layer_idx][:, :, t:t+1, :] = k_new
        self.v[layer_idx][:, :, t:t+1, :] = v_new

    def incr_t(self):
        self.t += 1


class Embedding:
    def __init__(self, B, num_embeddings: int, embedding_dim: int):
        self.B=B
        self.weight=B["Tensor"].randn((num_embeddings, embedding_dim), std=0.08, as_param=True)

    def forward_1d(self, idx: int):
        if self.B["kind"]=="list":
            return self.weight.row2d(idx)
        return self.weight.row(idx)

    def forward_batch(self, idx_bt):
        if self.B["kind"]=="list":
            return self.B["Tensor"].gather_rows_batch(self.weight, idx_bt)
        torch=self.B["torch"]
        return torch.embedding(self.weight.data, idx_bt)

    def state_dict(self):
        if self.B["kind"]=="torch":
            return {"shape": self.weight.shape, "weight": self.weight.data.detach().float().cpu().tolist()}
        return {"shape": self.weight.shape, "weight": self.B["pack_floats"](self.weight)}

    def load_state_dict(self, sd):
        if self.B["kind"]=="torch":
            self.B["unpack_param_like"](self.weight.data, sd["weight"])
        else:
            self.B["unpack_tensor_like"](self.weight, sd["weight"])


class Linear:
    def __init__(self, B, in_features: int, out_features: int):
        self.B=B
        self.weight=B["Tensor"].randn((out_features, in_features), std=0.08, as_param=True)

    def forward_1d(self, x):
        return self.B["Tensor"].matvec(self.weight, x)

    def forward_batch(self, x_bti):
        if self.B["kind"]=="list":
            return self.B["Tensor"].linear_btc(x_bti, self.weight)
        return x_bti @ self.weight.data.t()

    def state_dict(self):
        if self.B["kind"]=="torch":
            return {"shape": self.weight.shape, "weight": self.weight.data.detach().float().cpu().tolist()}
        return {"shape": self.weight.shape, "weight": self.B["pack_floats"](self.weight)}

    def load_state_dict(self, sd):
        if self.B["kind"]=="torch":
            self.B["unpack_param_like"](self.weight.data, sd["weight"])
        else:
            self.B["unpack_tensor_like"](self.weight, sd["weight"])


class RMSNorm:
    def __init__(self,B,n_embd,eps=1e-5):
        self.B=B; self.eps=float(eps)
        self.gamma=B["Tensor"].randn((n_embd,), std=0.02, as_param=True)

    def forward_1d(self,x):
        T=self.B["Tensor"]
        y=T.rmsnorm_1d(x,self.eps)
        return T.from_1d([y.data[i]*self.gamma.data[i] for i in range(self.gamma.shape[0])])

    def forward_batch(self,x_btc):
        if self.B["kind"]=="list":
            T=self.B["Tensor"]
            y=T.rmsnorm_btc(x_btc,self.eps)
            return T.mul_gamma_btc(y,self.gamma)
        torch=self.B["torch"]
        ms=(x_btc*x_btc).mean(dim=-1,keepdim=True)
        y=x_btc*torch.rsqrt(ms+self.eps)
        return y*self.gamma.data

    def state_dict(self):
        if self.B["kind"]=="torch":
            return {"eps": self.eps, "shape": self.gamma.shape, "gamma": self.gamma.data.detach().float().cpu().tolist()}
        return {"eps": self.eps, "shape": self.gamma.shape, "gamma": self.B["pack_floats"](self.gamma)}

    def load_state_dict(self, sd):
        self.eps=float(sd["eps"])
        if self.B["kind"]=="torch":
            self.B["unpack_param_like"](self.gamma.data, sd["gamma"])
        else:
            self.B["unpack_tensor_like"](self.gamma, sd["gamma"])


class CausalSelfAttention:
    def __init__(self,B,n_embd,n_head):
        self.B=B
        self.n_embd=int(n_embd); self.n_head=int(n_head)
        assert self.n_embd % self.n_head==0
        self.head_dim=self.n_embd//self.n_head
        self.scale=1.0/(self.head_dim**0.5)
        self.wq=Linear(B,n_embd,n_embd)
        self.wk=Linear(B,n_embd,n_embd)
        self.wv=Linear(B,n_embd,n_embd)
        self.wo=Linear(B,n_embd,n_embd)

    # list generation step with KV cache
    def forward_step_list(self,x,cache:KVCache,layer_idx:int):
        T=self.B["Tensor"]
        q=self.wq.forward_1d(x)
        k=self.wk.forward_1d(x)
        v=self.wv.forward_1d(x)

        qh=T.split_heads_1d(q,self.n_head)
        kh=T.split_heads_1d(k,self.n_head)
        vh=T.split_heads_1d(v,self.n_head)
        cache.append(layer_idx,kh,vh)

        out_heads=[]
        for h in range(self.n_head):
            q_h=qh[h]
            k_hist=cache.keys[layer_idx][h]
            v_hist=cache.values[layer_idx][h]
            TT=len(k_hist)

            logits=[]
            for t in range(TT):
                s=None
                for j in range(self.head_dim):
                    term=q_h.data[j]*k_hist[t].data[j]
                    s=term if s is None else (s+term)
                logits.append(s*self.scale)

            w=T.from_1d(logits).softmax_1d()

            out=[]
            for j in range(self.head_dim):
                s=None
                for t in range(TT):
                    term=w.data[t]*v_hist[t].data[j]
                    s=term if s is None else (s+term)
                out.append(s)
            out_heads.append(T.from_1d(out))

        y=T.cat_heads_1d(out_heads)
        return self.wo.forward_1d(y)

    # torch generation step with PREALLOCATED KV cache (batch)
    def forward_step_torch(self, x_bC, cache: TorchKVCachePrealloc, layer_idx: int):
        torch = self.B["torch"]
        B = x_bC.shape[0]
        H = self.n_head
        Hd = self.head_dim
        t = cache.t

        q = x_bC @ self.wq.weight.data.t()
        k = x_bC @ self.wk.weight.data.t()
        v = x_bC @ self.wv.weight.data.t()

        q = q.view(B, H, 1, Hd)
        k = k.view(B, H, 1, Hd)
        v = v.view(B, H, 1, Hd)

        cache.append(layer_idx, k, v)
        K = cache.k[layer_idx][:, :, :t+1, :]  # [B,H,T,Hd]
        V = cache.v[layer_idx][:, :, :t+1, :]

        att = (q @ K.transpose(-2, -1)) * self.scale  # [B,H,1,T]
        w = torch.softmax(att, dim=-1)
        y = w @ V                                    # [B,H,1,Hd]
        y = y.view(B, H*Hd)
        out = y @ self.wo.weight.data.t()
        return out

    def forward_batch(self, x_btc, causal_mask=None):
        if self.B["kind"]=="list":
            T=self.B["Tensor"]
            H=self.n_head

            q=self.wq.forward_batch(x_btc)  # [B,T,C]
            k=self.wk.forward_batch(x_btc)
            v=self.wv.forward_batch(x_btc)

            q=T.reshape_bthd(q,H)           # [B,H,T,Hd]
            k=T.reshape_bthd(k,H)
            v=T.reshape_bthd(v,H)
            kT=T.transpose_last2(k)         # [B,H,Hd,T]
            att=T.matmul_bhtt(q,kT)         # [B,H,T,T]
            att=T.scale_bhtt(att,self.scale)
            att=T.causal_mask_bhtt(att,mask_value=-1e9)
            w=T.softmax_last_bhtt(att)
            y=T.matmul_bhtd(w,v)            # [B,H,T,Hd]
            y=T.merge_heads_btc(y)          # [B,T,C]
            return self.wo.forward_batch(y) # [B,T,C]

        torch=self.B["torch"]
        Bsz,T,C=x_btc.shape
        H=self.n_head; Hd=self.head_dim
        q=self.wq.forward_batch(x_btc)
        k=self.wk.forward_batch(x_btc)
        v=self.wv.forward_batch(x_btc)
        q=q.view(Bsz,T,H,Hd).transpose(1,2)
        k=k.view(Bsz,T,H,Hd).transpose(1,2)
        v=v.view(Bsz,T,H,Hd).transpose(1,2)
        att=(q@k.transpose(-2,-1))*self.scale
        att=att.masked_fill(causal_mask==0,float("-inf"))
        w=torch.softmax(att,dim=-1)
        y=w@v
        y=y.transpose(1,2).contiguous().view(Bsz,T,C)
        return self.wo.forward_batch(y)

    def state_dict(self):
        return {"wq": self.wq.state_dict(),"wk": self.wk.state_dict(),"wv": self.wv.state_dict(),"wo": self.wo.state_dict()}

    def load_state_dict(self,sd):
        self.wq.load_state_dict(sd["wq"])
        self.wk.load_state_dict(sd["wk"])
        self.wv.load_state_dict(sd["wv"])
        self.wo.load_state_dict(sd["wo"])


class MLP:
    def __init__(self,B,n_embd):
        self.B=B
        self.fc1=Linear(B,n_embd,4*n_embd)
        self.fc2=Linear(B,4*n_embd,n_embd)

    def forward_batch(self,x_btc):
        if self.B["kind"]=="list":
            T=self.B["Tensor"]
            h=self.fc1.forward_batch(x_btc)
            h=T.relu_btc(h)
            return self.fc2.forward_batch(h)
        torch=self.B["torch"]
        h=self.fc1.forward_batch(x_btc)
        h=torch.relu(h)
        return self.fc2.forward_batch(h)

    def state_dict(self):
        return {"fc1": self.fc1.state_dict(),"fc2": self.fc2.state_dict()}

    def load_state_dict(self,sd):
        self.fc1.load_state_dict(sd["fc1"])
        self.fc2.load_state_dict(sd["fc2"])


class Block:
    def __init__(self,B,n_embd,n_head):
        self.B=B
        self.norm1=RMSNorm(B,n_embd)
        self.attn=CausalSelfAttention(B,n_embd,n_head)
        self.norm2=RMSNorm(B,n_embd)
        self.mlp=MLP(B,n_embd)

    def forward_batch(self,x_btc,causal_mask=None):
        if self.B["kind"]=="list":
            T=self.B["Tensor"]
            x = T.add_btc(x_btc, self.attn.forward_batch(self.norm1.forward_batch(x_btc)))
            x = T.add_btc(x, self.mlp.forward_batch(self.norm2.forward_batch(x)))
            return x
        x = x_btc + self.attn.forward_batch(self.norm1.forward_batch(x_btc), causal_mask)
        x = x + self.mlp.forward_batch(self.norm2.forward_batch(x))
        return x

    # list KV step for sampling
    def forward_step_list(self, x, cache:KVCache, layer_idx:int):
        T=self.B["Tensor"]
        a = self.attn.forward_step_list(self.norm1.forward_1d(x), cache, layer_idx)
        x = T.from_1d([x.data[i] + a.data[i] for i in range(len(x.data))])

        h = self.mlp.fc1.forward_1d(self.norm2.forward_1d(x))
        h = T.relu_1d(h)
        m = self.mlp.fc2.forward_1d(h)
        x = T.from_1d([x.data[i] + m.data[i] for i in range(len(x.data))])
        return x

    # torch KV step for sampling (batch) with prealloc cache
    def forward_step_torch(self, x_bC, cache: TorchKVCachePrealloc, layer_idx: int):
        torch = self.B["torch"]

        ms = (x_bC * x_bC).mean(dim=-1, keepdim=True)
        y = x_bC * torch.rsqrt(ms + self.norm1.eps)
        y = y * self.norm1.gamma.data

        a = self.attn.forward_step_torch(y, cache, layer_idx)
        x_bC = x_bC + a

        ms = (x_bC * x_bC).mean(dim=-1, keepdim=True)
        y = x_bC * torch.rsqrt(ms + self.norm2.eps)
        y = y * self.norm2.gamma.data

        h = y @ self.mlp.fc1.weight.data.t()
        h = torch.relu(h)
        m = h @ self.mlp.fc2.weight.data.t()
        x_bC = x_bC + m
        return x_bC

    def state_dict(self):
        return {"norm1": self.norm1.state_dict(),"attn": self.attn.state_dict(),"norm2": self.norm2.state_dict(),"mlp": self.mlp.state_dict()}

    def load_state_dict(self,sd):
        self.norm1.load_state_dict(sd["norm1"])
        self.attn.load_state_dict(sd["attn"])
        self.norm2.load_state_dict(sd["norm2"])
        self.mlp.load_state_dict(sd["mlp"])


class MicroGPT:
    def __init__(self,B,vocab_size,n_layer,n_embd,block_size,n_head):
        self.B=B
        self.vocab_size=int(vocab_size)
        self.n_layer=int(n_layer)
        self.n_embd=int(n_embd)
        self.block_size=int(block_size)
        self.n_head=int(n_head)

        self.wte=Embedding(B,self.vocab_size,self.n_embd)
        self.wpe=Embedding(B,self.block_size,self.n_embd)
        self.blocks=[Block(B,self.n_embd,self.n_head) for _ in range(self.n_layer)]
        self.norm_f=RMSNorm(B,self.n_embd)
        self.lm_head=Linear(B,self.n_embd,self.vocab_size)

    def named_parameters(self,prefix=""):
        if self.B["kind"]=="torch":
            yield (prefix+"wte.weight", self.wte.weight.data)
            yield (prefix+"wpe.weight", self.wpe.weight.data)
            for li,blk in enumerate(self.blocks):
                bp=f"{prefix}blocks.{li}."
                yield (bp+"norm1.gamma", blk.norm1.gamma.data)
                yield (bp+"attn.wq.weight", blk.attn.wq.weight.data)
                yield (bp+"attn.wk.weight", blk.attn.wk.weight.data)
                yield (bp+"attn.wv.weight", blk.attn.wv.weight.data)
                yield (bp+"attn.wo.weight", blk.attn.wo.weight.data)
                yield (bp+"norm2.gamma", blk.norm2.gamma.data)
                yield (bp+"mlp.fc1.weight", blk.mlp.fc1.weight.data)
                yield (bp+"mlp.fc2.weight", blk.mlp.fc2.weight.data)
            yield (prefix+"norm_f.gamma", self.norm_f.gamma.data)
            yield (prefix+"lm_head.weight", self.lm_head.weight.data)
        else:
            yield (prefix+"wte.weight", self.wte.weight.data)
            yield (prefix+"wpe.weight", self.wpe.weight.data)
            for li,blk in enumerate(self.blocks):
                bp=f"{prefix}blocks.{li}."
                yield (bp+"norm1.gamma", blk.norm1.gamma.data)
                yield (bp+"attn.wq.weight", blk.attn.wq.weight.data)
                yield (bp+"attn.wk.weight", blk.attn.wk.weight.data)
                yield (bp+"attn.wv.weight", blk.attn.wv.weight.data)
                yield (bp+"attn.wo.weight", blk.attn.wo.weight.data)
                yield (bp+"norm2.gamma", blk.norm2.gamma.data)
                yield (bp+"mlp.fc1.weight", blk.mlp.fc1.weight.data)
                yield (bp+"mlp.fc2.weight", blk.mlp.fc2.weight.data)
            yield (prefix+"norm_f.gamma", self.norm_f.gamma.data)
            yield (prefix+"lm_head.weight", self.lm_head.weight.data)

    # list KV step (sampling)
    def forward_step_list(self, token_id: int, pos_id: int, cache: KVCache):
        T=self.B["Tensor"]
        tok=self.wte.forward_1d(token_id)
        pos=self.wpe.forward_1d(pos_id)
        x=T.from_1d([tok.data[i]+pos.data[i] for i in range(self.n_embd)])
        for li,blk in enumerate(self.blocks):
            x = blk.forward_step_list(x, cache, li)
        x = self.norm_f.forward_1d(x)
        return self.lm_head.forward_1d(x)

    # torch KV step (sampling, batch) with prealloc cache
    def forward_step_torch(self, token_ids_b, pos_id: int, cache: TorchKVCachePrealloc):
        torch = self.B["torch"]
        B = token_ids_b.shape[0]
        C = self.n_embd
        tok = self.wte.weight.data[token_ids_b]  # [B,C]
        pos = self.wpe.weight.data[pos_id].view(1, C).expand(B, C)
        x = tok + pos

        for li,blk in enumerate(self.blocks):
            x = blk.forward_step_torch(x, cache, li)

        ms = (x * x).mean(dim=-1, keepdim=True)
        y = x * torch.rsqrt(ms + self.norm_f.eps)
        y = y * self.norm_f.gamma.data
        logits = y @ self.lm_head.weight.data.t()  # [B,V]
        return logits

    # training forward batch (both backends)
    def forward_batch(self, x_bT: List[List[int]]):
        if self.B["kind"]=="list":
            Tns=self.B["Tensor"]
            Bsz=len(x_bT); T=len(x_bT[0])
            tok=self.wte.forward_batch(x_bT)  # [B,T,C]
            pos_ids=[list(range(T)) for _ in range(Bsz)]
            pos=self.wpe.forward_batch(pos_ids)  # [B,T,C]
            h=Tns.add_btc(tok,pos)
            for blk in self.blocks:
                h=blk.forward_batch(h)
            h=self.norm_f.forward_batch(h)
            logits=self.lm_head.forward_batch(h)  # [B,T,V]
            return {"kind":"list_logits","B":Bsz,"T":T,"V":self.vocab_size,"data":logits.data}

        torch=self.B["torch"]
        device=self.B["device"]
        x=torch.tensor(x_bT,device=device,dtype=torch.long)
        Bsz,T=x.shape
        tok=self.wte.forward_batch(x)
        pos_ids=torch.arange(T,device=device,dtype=torch.long).unsqueeze(0).expand(Bsz,T)
        pos=self.wpe.forward_batch(pos_ids)
        h=tok+pos
        causal_mask=torch.tril(torch.ones(T,T,device=device,dtype=torch.bool)).view(1,1,T,T)
        for blk in self.blocks:
            h=blk.forward_batch(h,causal_mask)
        h=self.norm_f.forward_batch(h)
        logits=self.lm_head.forward_batch(h)
        return logits

    def state_dict(self):
        return {
            "wte": self.wte.state_dict(),
            "wpe": self.wpe.state_dict(),
            "blocks": [b.state_dict() for b in self.blocks],
            "norm_f": self.norm_f.state_dict(),
            "lm_head": self.lm_head.state_dict(),
            "meta": {"vocab_size": self.vocab_size, "n_layer": self.n_layer, "n_embd": self.n_embd, "block_size": self.block_size, "n_head": self.n_head},
        }

    def load_state_dict(self, sd):
        self.wte.load_state_dict(sd["wte"])
        self.wpe.load_state_dict(sd["wpe"])
        for b, bs in zip(self.blocks, sd["blocks"]):
            b.load_state_dict(bs)
        self.norm_f.load_state_dict(sd["norm_f"])
        self.lm_head.load_state_dict(sd["lm_head"])


# ============================================================
# Loss / schedule / sampling
# ============================================================
def lr_scale_warmup_linear(step: int, total_steps: int, warmup_steps: int) -> float:
    if total_steps <= 0: return 1.0
    if warmup_steps > 0 and step <= warmup_steps:
        return step / warmup_steps
    remaining = max(0, total_steps - step)
    denom = max(1, total_steps - warmup_steps)
    return remaining / denom


def loss_batch(B, logits, y_bT: List[List[int]]):
    if B["kind"] == "torch":
        torch = B["torch"]
        device = B["device"]
        y = torch.tensor(y_bT, device=device, dtype=torch.long)
        Bsz, T, V = logits.shape
        return torch.nn.functional.cross_entropy(logits.reshape(Bsz * T, V), y.reshape(Bsz * T), reduction="mean")

    Value = B["Value"]
    Tensor = B["Tensor"]
    Bsz, T = logits["B"], logits["T"]
    total = Value(0.0)
    n = Bsz * T
    for b in range(Bsz):
        for t in range(T):
            probs = Tensor.from_1d(logits["data"][b][t]).softmax_1d()
            total = total + (-probs.data[int(y_bT[b][t])].log())
    return (1 / n) * total


def sample_next_token_from_probs(probs: List[float], top_k: int, top_p: float) -> int:
    cand = list(enumerate(probs))
    cand.sort(key=lambda x: x[1], reverse=True)
    if top_k and top_k > 0:
        cand = cand[:top_k]
    if top_p and 0.0 < top_p < 1.0:
        kept, cum = [], 0.0
        for idx, p in cand:
            kept.append((idx, p)); cum += p
            if cum >= top_p: break
        cand = kept
    total = sum(p for _, p in cand)
    if total <= 0: return cand[0][0]
    weights = [p / total for _, p in cand]
    ids = [i for i, _ in cand]
    return random.choices(ids, weights=weights, k=1)[0]


def sample_next_token_list(B, logits_v, temperature: float, top_k: int, top_p: float) -> int:
    if temperature <= 0: temperature = 1e-6
    scaled = [v / temperature for v in logits_v.data]
    probs = B["Tensor"].from_1d(scaled).softmax_1d()
    probs = [p.data for p in probs.data]
    return sample_next_token_from_probs(probs, top_k, top_p)


# ============================================================
# Checkpoint
# ============================================================
def save_ckpt(path: str, step: int, cfg: Config, tok: CharTokenizer, model: MicroGPT, opt_state: dict, backend_kind: str, extra: dict):
    ckpt = {
        "step": int(step),
        "cfg": asdict(cfg),
        "backend_kind": backend_kind,
        "tokenizer": tok.state_dict(),
        "model": model.state_dict(),
        "opt_unified_state": opt_state,
        "rng_state": random.getstate(),
        "extra": extra,
    }
    with open(path, "wb") as f:
        pickle.dump(ckpt, f)


def load_ckpt(path: str) -> dict:
    with open(path, "rb") as f:
        return pickle.load(f)


# ============================================================
# Sampling
# ============================================================
def generate_samples_list_kv(B, model: MicroGPT, tok: CharTokenizer, n_samples: int, temperature: float, top_k: int, top_p: float):
    print("\n--- inference samples (list KV) ---")
    for i in range(n_samples):
        cache = KVCache(model.n_layer, model.n_head)
        token_id = tok.BOS
        out=[]
        with B["NoGrad"]():
            for pos_id in range(model.block_size):
                logits_v = model.forward_step_list(token_id, pos_id, cache)  # Tensor[ V ]
                token_id = sample_next_token_list(B, logits_v, temperature, top_k, top_p)
                if token_id == tok.BOS:
                    break
                out.append(tok.uchars[token_id])
        print(f"sample {i+1:2d}: {''.join(out)}")


def generate_samples_torch_kv_prealloc(B, model: MicroGPT, tok: CharTokenizer, n_samples: int, temperature: float, top_k: int, top_p: float):
    print("\n--- inference samples (torch KV batch prealloc) ---")
    torch = B["torch"]
    device = B["device"]
    maxT = model.block_size
    BOS = tok.BOS
    H = model.n_head
    Hd = model.n_embd // model.n_head

    dtype = model.wte.weight.data.dtype
    cache = TorchKVCachePrealloc(torch, device, model.n_layer, H, Hd, n_samples, maxT, dtype)

    token_ids = torch.full((n_samples,), BOS, device=device, dtype=torch.long)
    finished = torch.zeros((n_samples,), device=device, dtype=torch.bool)
    outputs = [[] for _ in range(n_samples)]

    with torch.no_grad():
        for pos_id in range(maxT):
            logits = model.forward_step_torch(token_ids, pos_id, cache)  # [B,V]
            scaled = logits / max(temperature, 1e-6)
            probs = torch.softmax(scaled, dim=-1).detach().cpu().tolist()

            next_ids = []
            for b in range(n_samples):
                if finished[b].item():
                    next_ids.append(BOS)
                    continue
                nid = sample_next_token_from_probs(probs[b], top_k, top_p)
                next_ids.append(nid)
                if nid == BOS:
                    finished[b] = True
                else:
                    outputs[b].append(tok.uchars[nid])

            token_ids = torch.tensor(next_ids, device=device, dtype=torch.long)
            cache.incr_t()
            if bool(finished.all()):
                break

    for i in range(n_samples):
        print(f"sample {i+1:2d}: {''.join(outputs[i])}")

# Main
# ============================================================
def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--backend", choices=["list", "torch"], default="list")
    ap.add_argument("--device", default="cpu")
    ap.add_argument("--resume", action="store_true")
    args = ap.parse_args()

    cfg = Config()
    docs = load_docs(cfg.input_path, cfg.fallback_url)
    tok = CharTokenizer(docs)
    encoded = [tok.encode_with_bos(d) for d in docs]
    sampler = NameSampler(encoded, cfg.block_size, cfg.random_offset)

    B = build_backend_torch(args.device) if args.backend == "torch" else build_backend_list()
    model = MicroGPT(B, tok.vocab_size, cfg.n_layer, cfg.n_embd, cfg.block_size, cfg.n_head)

    named_params = list(model.named_parameters())
    opt = AdamUnifiedNamedTorch(B, named_params, cfg.learning_rate, cfg.beta1, cfg.beta2, cfg.eps_adam) if B["kind"]=="torch" \
          else AdamUnifiedNamedList(B, named_params, cfg.learning_rate, cfg.beta1, cfg.beta2, cfg.eps_adam)

    start_step = 0
    if args.resume and os.path.exists(cfg.ckpt_path):
        ckpt = load_ckpt(cfg.ckpt_path)
        start_step = int(ckpt.get("step", 0))
        tok = CharTokenizer.load_state_dict(ckpt["tokenizer"])
        model = MicroGPT(B, tok.vocab_size, cfg.n_layer, cfg.n_embd, cfg.block_size, cfg.n_head)
        model.load_state_dict(ckpt["model"])
        named_params = list(model.named_parameters())
        opt = AdamUnifiedNamedTorch(B, named_params, cfg.learning_rate, cfg.beta1, cfg.beta2, cfg.eps_adam) if B["kind"]=="torch" \
              else AdamUnifiedNamedList(B, named_params, cfg.learning_rate, cfg.beta1, cfg.beta2, cfg.eps_adam)
        if "rng_state" in ckpt:
            random.setstate(ckpt["rng_state"])
        ustate = ckpt.get("opt_unified_state")
        if ustate:
            opt.load_unified_state(ustate)

    print(f"\nbackend={B['kind']} device={args.device} vocab={tok.vocab_size} steps={cfg.num_steps} batch={cfg.batch_size}")

    for step0 in range(start_step, cfg.num_steps):
        step = step0 + 1
        x_bT, y_bT = sampler.next_batch(cfg.batch_size)

        logits = model.forward_batch(x_bT)
        loss = loss_batch(B, logits, y_bT)

        if B["kind"] == "torch":
            torch = B["torch"]
            opt.zero_grad()
            loss.backward()
            if cfg.grad_clip_norm and cfg.grad_clip_norm > 0:
                torch.nn.utils.clip_grad_norm_([p for _, p in named_params], cfg.grad_clip_norm)
            lr_scale = lr_scale_warmup_linear(step, cfg.num_steps, cfg.warmup_steps)
            opt.step(lr_scale)
            loss_val = float(loss.detach().cpu())
        else:
            loss.backward()
            if cfg.grad_clip_norm and cfg.grad_clip_norm > 0:
                leaves=[]
                def iter_leaf(x):
                    if hasattr(x, "grad") and hasattr(x, "data"):
                        yield x
                    elif isinstance(x, list):
                        for it in x: yield from iter_leaf(it)
                for _, root in named_params:
                    leaves.extend(list(iter_leaf(root)))
                B["clip_grad_norm"](leaves, cfg.grad_clip_norm)
            lr_scale = lr_scale_warmup_linear(step, cfg.num_steps, cfg.warmup_steps)
            opt.step(lr_scale)
            loss_val = loss.data

        print(f"step {step:4d}/{cfg.num_steps:4d} | loss {loss_val:.4f}", end="\r")

        if cfg.save_every and cfg.save_every > 0 and step % cfg.save_every == 0:
            save_ckpt(cfg.ckpt_path, step, cfg, tok, model, opt.unified_state(), B["kind"], {"device": args.device})

    print("\ntraining done.")
    save_ckpt(cfg.ckpt_path, cfg.num_steps, cfg, tok, model, opt.unified_state(), B["kind"], {"device": args.device})
    print(f"saved: {cfg.ckpt_path}")

    if B["kind"] == "list":
        generate_samples_list_kv(B, model, tok, cfg.n_samples, cfg.temperature, cfg.top_k, cfg.top_p)
    else:
        generate_samples_torch_kv_prealloc(B, model, tok, cfg.n_samples, cfg.temperature, cfg.top_k, cfg.top_p)
        # 想用 list KV 采样：直接 --backend list --resume 复用同一 ckpt


if __name__ == "__main__":
    main()
