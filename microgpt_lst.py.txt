# ===================== SECTION: Utilities =====================

def prod(xs):
    p = 1
    for x in xs: p *= x
    return p

def clamp(a, lo, hi):
    return lo if a < lo else hi if a > hi else a

def assert_(cond, msg="assert failed"):
    if not cond:
        raise AssertionError(msg)

# ===================== SECTION: Tensor (pure Python, array('f')) =====================

class Tensor:
    """
    Pure Python float32 tensor backed by array('f') with shape/strides/offset.
    Supports view/slice/stack; non-contiguous ops will conservatively materialize contiguous() in policy A.
    """
    __slots__ = ("data", "shape", "strides", "offset", "requires_grad", "grad",
                 "_base", "_is_view", "_name")

    def __init__(self, data, shape, strides=None, offset=0, requires_grad=False, grad=None, base=None, is_view=False, name=""):
        self.data = data  # array('f')
        self.shape = tuple(shape)
        if strides is None:
            strides = Tensor.default_strides(self.shape)
        self.strides = tuple(strides)
        self.offset = int(offset)
        self.requires_grad = bool(requires_grad)
        self.grad = grad  # Tensor or None
        self._base = base
        self._is_view = is_view
        self._name = name

    @staticmethod
    def default_strides(shape):
        # row-major contiguous strides in elements
        st = []
        acc = 1
        for s in reversed(shape):
            st.append(acc)
            acc *= s
        return tuple(reversed(st))

    @staticmethod
    def zeros(shape, requires_grad=False, name=""):
        n = prod(shape)
        buf = array('f', [0.0]) * n
        grad = None
        if requires_grad:
            gbuf = array('f', [0.0]) * n
            grad = Tensor(gbuf, shape, requires_grad=False, name=name+".grad")
        return Tensor(buf, shape, requires_grad=requires_grad, grad=grad, name=name)

    @staticmethod
    def randn(shape, std=0.02, requires_grad=False, name=""):
        n = prod(shape)
        buf = array('f', [0.0]) * n
        for i in range(n):
            buf[i] = random.gauss(0.0, std)
        grad = None
        if requires_grad:
            gbuf = array('f', [0.0]) * n
            grad = Tensor(gbuf, shape, requires_grad=False, name=name+".grad")
        return Tensor(buf, shape, requires_grad=requires_grad, grad=grad, name=name)

    def numel(self):
        return prod(self.shape)

    def is_contiguous(self):
        return self.strides == Tensor.default_strides(self.shape) and self.offset == 0 and (self._base is None)

    def contiguous(self):
        if self.is_contiguous():
            return self
        out = Tensor.zeros(self.shape, requires_grad=self.requires_grad, name=self._name + ".contig")
        # copy elementwise
        it = NDIndexer(self.shape)
        while it.next():
            out.data[it.flat] = self.data[self._offset_index(it.idx)]
        # NOTE: grad buffer policy: leaf params carry grad; intermediates typically not.
        return out

    def _offset_index(self, idx):
        # idx tuple -> underlying data index
        o = self.offset
        for i, s in zip(idx, self.strides):
            o += i * s
        return o

    def view(self, new_shape):
        # only allow view if contiguous and same numel
        assert_(prod(new_shape) == self.numel(), "view: numel mismatch")
        if not self.is_contiguous():
            # policy A: materialize
            if Config.grad_view_policy == "A":
                return self.contiguous().view(new_shape)
            # policy B: allow view via base (still needs correct grad mapping)
        return Tensor(self.data, new_shape, strides=Tensor.default_strides(new_shape),
                      offset=self.offset, requires_grad=self.requires_grad,
                      grad=None, base=self if self._base is None else self._base,
                      is_view=True, name=self._name + ".view")

    def slice(self, slices):
        """
        Full multi-dim slicing with step.
        slices: list of slice/int/None, length <= ndim. None means full.
        Returns view if possible (strided), may be non-contiguous.
        """
        ndim = len(self.shape)
        sl = list(slices) + [None] * (ndim - len(slices))
        new_shape = []
        new_strides = []
        new_offset = self.offset
        for d in range(ndim):
            s = sl[d]
            if s is None:
                new_shape.append(self.shape[d])
                new_strides.append(self.strides[d])
            elif isinstance(s, int):
                # reduce dimension
                i = s
                if i < 0: i += self.shape[d]
                assert_(0 <= i < self.shape[d], "slice: index out of range")
                new_offset += i * self.strides[d]
            elif isinstance(s, slice):
                start, stop, step = s.start, s.stop, s.step
                if step is None: step = 1
                dim = self.shape[d]
                if start is None: start = 0
                if stop is None: stop = dim
                if start < 0: start += dim
                if stop < 0: stop += dim
                start = clamp(start, 0, dim)
                stop = clamp(stop, 0, dim)
                length = 0
                if step > 0:
                    if stop > start:
                        length = (stop - start + step - 1) // step
                else:
                    # negative step
                    if stop < start:
                        length = (start - stop + (-step) - 1) // (-step)
                new_offset += start * self.strides[d]
                new_shape.append(length)
                new_strides.append(self.strides[d] * step)
            else:
                raise TypeError("slice: invalid spec")
        out = Tensor(self.data, tuple(new_shape), strides=tuple(new_strides), offset=new_offset,
                     requires_grad=self.requires_grad, grad=None,
                     base=self if self._base is None else self._base, is_view=True, name=self._name + ".slice")
        # policy A: materialize if requires grad and view would need grad mapping later
        if Config.grad_view_policy == "A" and out.requires_grad and (not out.is_contiguous()):
            return out.contiguous()
        return out

    @staticmethod
    def stack(tensors, dim=0, requires_grad=False, name="stack"):
        assert_(len(tensors) > 0, "stack: empty")
        ref = tensors[0]
        for t in tensors:
            assert_(t.shape == ref.shape, "stack: shapes must match")
        ndim = len(ref.shape)
        dim = dim if dim >= 0 else dim + (ndim + 1)
        assert_(0 <= dim <= ndim, "stack: dim out of range")
        out_shape = list(ref.shape)
        out_shape.insert(dim, len(tensors))
        out = Tensor.zeros(tuple(out_shape), requires_grad=requires_grad, name=name)
        # copy
        it = NDIndexer(ref.shape)
        while it.next():
            for i, t in enumerate(tensors):
                out_idx = list(it.idx)
                out_idx.insert(dim, i)
                out.data[out._offset_index(tuple(out_idx))] = t.data[t._offset_index(it.idx)]
        return out

class NDIndexer:
    __slots__ = ("shape", "idx", "flat", "_done")
    def __init__(self, shape):
        self.shape = tuple(shape)
        self.idx = [0] * len(self.shape)
        self.flat = -1
        self._done = False

    def next(self):
        if self._done: return False
        if self.flat == -1:
            self.flat = 0
            return True
        # increment
        for d in reversed(range(len(self.shape))):
            self.idx[d] += 1
            if self.idx[d] < self.shape[d]:
                break
            self.idx[d] = 0
            if d == 0:
                self._done = True
                return False
        self.flat += 1
        return True

# ===================== SECTION: Tape (non-recursive autograd) =====================

class Op:
    __slots__ = ("name", "ctx", "bwd")
    def __init__(self, name, ctx, bwd):
        self.name = name
        self.ctx = ctx
        self.bwd = bwd  # function(ctx)

class Tape:
    __slots__ = ("ops",)
    def __init__(self):
        self.ops = []

    def add(self, name, ctx, bwd):
        self.ops.append(Op(name, ctx, bwd))

    def backward(self, loss):
        assert_(loss.grad is not None, "loss must have grad buffer")
        # seed dL/dL = 1
        loss.grad.data[0] = 1.0
        for op in reversed(self.ops):
            op.bwd(op.ctx)

# ===================== SECTION: PyKernels (fused kernels live here) =====================

class PyKernels:

    # ----- basic elementwise -----
    @staticmethod
    def add_fwd(a, b, out):
        it = NDIndexer(out.shape)
        while it.next():
            out.data[it.flat] = a.data[a._offset_index(tuple(it.idx))] + b.data[b._offset_index(tuple(it.idx))]

    @staticmethod
    def add_bwd(a, b, out):
        # out.grad -> a.grad, b.grad
        it = NDIndexer(out.shape)
        while it.next():
            go = out.grad.data[it.flat]
            if a.grad is not None:
                a.grad.data[a._offset_index(tuple(it.idx))] += go
            if b.grad is not None:
                b.grad.data[b._offset_index(tuple(it.idx))] += go

    @staticmethod
    def relu_fwd(x, out, mask):
        # mask: 1 if >0 else 0
        n = out.numel()
        for i in range(n):
            v = x.data[x.offset + i] if x.is_contiguous() else x.data[x._offset_index(_flat_to_idx(i, x.shape))]
            if v > 0.0:
                out.data[i] = v
                mask.data[i] = 1.0
            else:
                out.data[i] = 0.0
                mask.data[i] = 0.0

    @staticmethod
    def relu_bwd(x, out, mask):
        n = out.numel()
        for i in range(n):
            go = out.grad.data[i]
            gx = go * mask.data[i]
            if x.grad is not None:
                if x.is_contiguous():
                    x.grad.data[x.offset + i] += gx
                else:
                    x.grad.data[x._offset_index(_flat_to_idx(i, x.shape))] += gx

    # ----- embedding -----
    @staticmethod
    def embedding_fwd(W, idx_bt, out):
        # W: [V,C], idx_bt: [B,T] int list[list[int]], out: [B,T,C]
        B, T, C = out.shape
        for b in range(B):
            for t in range(T):
                wi = idx_bt[b][t]
                woff = wi * C
                ooff = (b*T + t) * C
                for c in range(C):
                    out.data[ooff + c] = W.data[woff + c]

    @staticmethod
    def embedding_bwd(W, idx_bt, out):
        # scatter-add into W.grad
        assert_(W.grad is not None, "W must have grad")
        B, T, C = out.shape
        for b in range(B):
            for t in range(T):
                wi = idx_bt[b][t]
                woff = wi * C
                ooff = (b*T + t) * C
                for c in range(C):
                    W.grad.data[woff + c] += out.grad.data[ooff + c]

    # ----- linear: Y = X @ W^T -----
    @staticmethod
    def linear_fwd(X, W, out):
        # X: [B,T,C], W: [O,C], out: [B,T,O]
        B, T, C = X.shape
        O, C2 = W.shape
        assert_(C == C2, "linear_fwd dim mismatch")
        for b in range(B):
            for t in range(T):
                xoff = (b*T + t) * C
                yoff = (b*T + t) * O
                for o in range(O):
                    woff = o * C
                    s = 0.0
                    for c in range(C):
                        s += X.data[xoff + c] * W.data[woff + c]
                    out.data[yoff + o] = s

    @staticmethod
    def linear_bwd(X, W, out):
        # dX += dY @ W, dW += dY^T @ X
        B, T, C = X.shape
        O, C2 = W.shape
        assert_(C == C2, "linear_bwd dim mismatch")
        for b in range(B):
            for t in range(T):
                xoff = (b*T + t) * C
                yoff = (b*T + t) * O
                for o in range(O):
                    go = out.grad.data[yoff + o]
                    woff = o * C
                    if X.grad is not None:
                        for c in range(C):
                            X.grad.data[xoff + c] += go * W.data[woff + c]
                    if W.grad is not None:
                        for c in range(C):
                            W.grad.data[woff + c] += go * X.data[xoff + c]

    # ----- rmsnorm -----
    @staticmethod
    def rmsnorm_fwd(X, out, inv_rms, eps=1e-5):
        # per (B,T) normalize across C
        B, T, C = X.shape
        for bt in range(B*T):
            xoff = bt * C
            ms = 0.0
            for c in range(C):
                v = X.data[xoff + c]
                ms += v * v
            ms /= C
            inv = 1.0 / math.sqrt(ms + eps)
            inv_rms.data[bt] = inv
            for c in range(C):
                out.data[xoff + c] = X.data[xoff + c] * inv

    @staticmethod
    def rmsnorm_bwd(X, out, inv_rms):
        # y = x * inv, inv = (mean(x^2)+eps)^-1/2
        # dL/dx_i = inv * g_i - (inv^3 / C) * x_i * sum_j(g_j * x_j)
        B, T, C = X.shape
        for bt in range(B*T):
            xoff = bt * C
            inv = inv_rms.data[bt]
            inv3 = inv * inv * inv
            dot = 0.0
            for c in range(C):
                dot += out.grad.data[xoff + c] * X.data[xoff + c]
            coef = (inv3 * dot) / C
            for c in range(C):
                gx = inv * out.grad.data[xoff + c] - coef * X.data[xoff + c]
                if X.grad is not None:
                    X.grad.data[xoff + c] += gx

    # ----- fused FFN: fc1 + relu + fc2 -----
    @staticmethod
    def ffn_fwd(X, W1, W2, out, act, mask):
        # X:[B,T,C], W1:[4C,C], act:[B,T,4C], mask same, W2:[C,4C], out:[B,T,C]
        PyKernels.linear_fwd(X, W1, act)
        # relu
        n = act.numel()
        for i in range(n):
            v = act.data[i]
            if v > 0.0:
                mask.data[i] = 1.0
            else:
                act.data[i] = 0.0
                mask.data[i] = 0.0
        PyKernels.linear_fwd(act, W2, out)

    @staticmethod
    def ffn_bwd(X, W1, W2, out, act, mask):
        # back through fc2, relu, fc1
        PyKernels.linear_bwd(act, W2, out)
        # relu backward on act (act is output of relu already; mask indicates where pre-activation>0)
        n = act.numel()
        for i in range(n):
            act.grad.data[i] *= mask.data[i]
        PyKernels.linear_bwd(X, W1, act)

    # ----- attention: training full-seq (scheme=2), stable softmax, workspace reuse -----
    @staticmethod
    def attn_train_fwd(Q, K, V, out, ws, n_head, head_dim):
        # Q,K,V: [B,T,C], out:[B,T,C]
        # ws fields (preallocated): logits[B,H,T,T], probs[B,H,T,T], dlogits[B,H,T,T], tmp[B,H,T,head_dim]
        B, T, C = Q.shape
        H = n_head
        D = head_dim
        scale = 1.0 / math.sqrt(D)

        # compute per head
        for b in range(B):
            for h in range(H):
                # logits[t, tp] = (q_t dot k_tp)*scale with causal mask
                for t in range(T):
                    # compute max for stable softmax with mask
                    maxv = -1e30
                    for tp in range(T):
                        if tp > t:
                            continue
                        s = 0.0
                        qoff = (b*T + t) * C + h*D
                        koff = (b*T + tp) * C + h*D
                        for i in range(D):
                            s += Q.data[qoff + i] * K.data[koff + i]
                        s *= scale
                        ws.logits[b][h][t*T + tp] = s
                        if s > maxv: maxv = s
                    # softmax denom
                    denom = 0.0
                    for tp in range(T):
                        if tp > t:
                            ws.probs[b][h][t*T + tp] = 0.0
                            continue
                        e = math.exp(ws.logits[b][h][t*T + tp] - maxv)
                        ws.probs[b][h][t*T + tp] = e
                        denom += e
                    invd = 1.0 / denom
                    for tp in range(T):
                        ws.probs[b][h][t*T + tp] *= invd
                # weighted sum: out[b,t,h*D:(h+1)D]
                for t in range(T):
                    ooff = (b*T + t) * C + h*D
                    for i in range(D):
                        s = 0.0
                        for tp in range(T):
                            if tp > t: continue
                            p = ws.probs[b][h][t*T + tp]
                            voff = (b*T + tp) * C + h*D
                            s += p * V.data[voff + i]
                        out.data[ooff + i] = s

    @staticmethod
    def attn_train_bwd(Q, K, V, out, ws, n_head, head_dim):
        # compute dV, dQ, dK using ws.probs and ws.dlogits workspace
        B, T, C = Q.shape
        H = n_head
        D = head_dim
        scale = 1.0 / math.sqrt(D)

        # dV: for each tp, i: sum_t p[t,tp] * go[t,i]
        for b in range(B):
            for h in range(H):
                for tp in range(T):
                    voff = (b*T + tp) * C + h*D
                    for i in range(D):
                        acc = 0.0
                        for t in range(T):
                            if tp > t: continue
                            p = ws.probs[b][h][t*T + tp]
                            go = out.grad.data[(b*T + t) * C + h*D + i]
                            acc += p * go
                        if V.grad is not None:
                            V.grad.data[voff + i] += acc

        # dprobs = go dot V
        # dlogits = p * (dprobs - sum(dprobs*p))
        for b in range(B):
            for h in range(H):
                for t in range(T):
                    # compute dprobs[tp]
                    sump = 0.0
                    for tp in range(T):
                        if tp > t:
                            ws.dlogits[b][h][t*T + tp] = 0.0
                            continue
                        # dot(go_t, v_tp)
                        dpr = 0.0
                        for i in range(D):
                            go = out.grad.data[(b*T + t)*C + h*D + i]
                            v = V.data[(b*T + tp)*C + h*D + i]
                            dpr += go * v
                        p = ws.probs[b][h][t*T + tp]
                        ws.dlogits[b][h][t*T + tp] = dpr  # temporarily store dprobs
                        sump += dpr * p
                    for tp in range(T):
                        if tp > t: continue
                        dpr = ws.dlogits[b][h][t*T + tp]
                        p = ws.probs[b][h][t*T + tp]
                        ws.dlogits[b][h][t*T + tp] = p * (dpr - sump)  # now dlogits

        # dQ += sum_tp dlogits[t,tp] * K[tp] * scale
        # dK += sum_t dlogits[t,tp] * Q[t] * scale
        for b in range(B):
            for h in range(H):
                for t in range(T):
                    qoff = (b*T + t) * C + h*D
                    for i in range(D):
                        acc = 0.0
                        for tp in range(T):
                            if tp > t: continue
                            dl = ws.dlogits[b][h][t*T + tp]
                            k = K.data[(b*T + tp)*C + h*D + i]
                            acc += dl * k
                        if Q.grad is not None:
                            Q.grad.data[qoff + i] += acc * scale
                for tp in range(T):
                    koff = (b*T + tp) * C + h*D
                    for i in range(D):
                        acc = 0.0
                        for t in range(T):
                            if tp > t: continue
                            dl = ws.dlogits[b][h][t*T + tp]
                            q = Q.data[(b*T + t)*C + h*D + i]
                            acc += dl * q
                        if K.grad is not None:
                            K.grad.data[koff + i] += acc * scale

    # ----- attention: inference step kernel with KV cache (fused softmax+weighted sum) -----
    @staticmethod
    def attn_step_fwd(q_bh, Kcache, Vcache, t, out_bh, tmp, Tmax, head_dim):
        # q_bh: [H,D] contiguous buffer (array)
        # Kcache/Vcache: layout (Tmax, H, B, D) but we will access for one b at a time: K[t',h,b,i]
        # out_bh: [H,D] contiguous
        # tmp: workspace logits/probs length t+1 per head (Python list reused)
        # stable softmax + weighted sum
        H = len(q_bh) // head_dim
        D = head_dim
        scale = 1.0 / math.sqrt(D)
        for h in range(H):
            # logits
            maxv = -1e30
            for tp in range(t+1):
                s = 0.0
                for i in range(D):
                    s += q_bh[h*D+i] * Kcache.get(tp, h, i)
                s *= scale
                tmp.logits[h][tp] = s
                if s > maxv: maxv = s
            denom = 0.0
            for tp in range(t+1):
                e = math.exp(tmp.logits[h][tp] - maxv)
                tmp.probs[h][tp] = e
                denom += e
            invd = 1.0 / denom
            for tp in range(t+1):
                tmp.probs[h][tp] *= invd
            # weighted sum
            for i in range(D):
                acc = 0.0
                for tp in range(t+1):
                    acc += tmp.probs[h][tp] * Vcache.get(tp, h, i)
                out_bh[h*D+i] = acc

    # ----- cross entropy: fused logsoftmax + NLL -----
    @staticmethod
    def logsoftmax_nll_fwd(logits, targets_bt, out_loss_bt, ws):
        # logits: [B,T,V], targets_bt: list[list[int]], out_loss_bt:[B,T]
        B, T, V = logits.shape
        # ws: per (B,T): max, lse
        for b in range(B):
            for t in range(T):
                off = (b*T + t)*V
                # max
                m = -1e30
                for v in range(V):
                    x = logits.data[off+v]
                    if x > m: m = x
                # lse
                s = 0.0
                for v in range(V):
                    s += math.exp(logits.data[off+v] - m)
                lse = m + math.log(s)
                ws.maxv[b*T+t] = m
                ws.lse[b*T+t] = lse
                y = targets_bt[b][t]
                out_loss_bt.data[b*T+t] = lse - logits.data[off+y]

    @staticmethod
    def logsoftmax_nll_bwd(logits, targets_bt, out_loss_bt, ws):
        # dlogits = softmax - onehot(target) scaled by out_loss_bt.grad (usually 1/(B*T))
        B, T, V = logits.shape
        for b in range(B):
            for t in range(T):
                g = out_loss_bt.grad.data[b*T+t]
                off = (b*T + t)*V
                m = ws.maxv[b*T+t]
                lse = ws.lse[b*T+t]
                # softmax
                for v in range(V):
                    p = math.exp(logits.data[off+v] - lse)  # stable
                    if logits.grad is not None:
                        logits.grad.data[off+v] += g * p
                y = targets_bt[b][t]
                if logits.grad is not None:
                    logits.grad.data[off+y] -= g

def _flat_to_idx(flat, shape):
    idx = [0]*len(shape)
    for d in reversed(range(len(shape))):
        s = shape[d]
        idx[d] = flat % s
        flat //= s
    return tuple(idx)

class Backend:
    # standard ops: create tensors, fused kernels, etc.
    def tensor_zeros(self, shape, requires_grad=False, name=""): raise NotImplementedError
    def tensor_randn(self, shape, std=0.02, requires_grad=False, name=""): raise NotImplementedError

    # ops
    def add(self, tape, a, b, name="add"): raise NotImplementedError
    def rmsnorm(self, tape, x, eps=1e-5, name="rmsnorm"): raise NotImplementedError
    def linear(self, tape, x, w, name="linear"): raise NotImplementedError
    def embedding(self, tape, W, idx_bt, name="embedding"): raise NotImplementedError
    def ffn(self, tape, x, w1, w2, name="ffn"): raise NotImplementedError

    # attention
    def attn_train(self, tape, q, k, v, n_head, head_dim, ws, name="attn_train"): raise NotImplementedError

    # cross entropy
    def cross_entropy(self, tape, logits, targets_bt, ws, name="cross_entropy"): raise NotImplementedError

class PyBackend(Backend):
    def tensor_zeros(self, shape, requires_grad=False, name=""):
        return Tensor.zeros(shape, requires_grad=requires_grad, name=name)
    def tensor_randn(self, shape, std=0.02, requires_grad=False, name=""):
        return Tensor.randn(shape, std=std, requires_grad=requires_grad, name=name)

    def add(self, tape, a, b, name="add"):
        assert_(a.shape == b.shape, "add shape mismatch")
        out = Tensor.zeros(a.shape, requires_grad=(a.requires_grad or b.requires_grad), name=name)
        if out.requires_grad:
            out.grad = Tensor.zeros(out.shape, requires_grad=False, name=name+".grad")
        PyKernels.add_fwd(a, b, out)
        def bwd(ctx):
            PyKernels.add_bwd(ctx["a"], ctx["b"], ctx["out"])
        tape.add(name, {"a":a, "b":b, "out":out}, bwd)
        return out

    def linear(self, tape, x, w, name="linear"):
        # x:[B,T,C], w:[O,C] => out:[B,T,O]
        B,T,C = x.shape
        O,C2 = w.shape
        assert_(C==C2, "linear dims")
        out = Tensor.zeros((B,T,O), requires_grad=(x.requires_grad or w.requires_grad), name=name)
        if out.requires_grad: out.grad = Tensor.zeros(out.shape, name=name+".grad")
        PyKernels.linear_fwd(x, w, out)
        def bwd(ctx):
            PyKernels.linear_bwd(ctx["x"], ctx["w"], ctx["out"])
        tape.add(name, {"x":x, "w":w, "out":out}, bwd)
        return out

    def embedding(self, tape, W, idx_bt, name="embedding"):
        B = len(idx_bt); T = len(idx_bt[0]); V,C = W.shape
        out = Tensor.zeros((B,T,C), requires_grad=W.requires_grad, name=name)
        if out.requires_grad: out.grad = Tensor.zeros(out.shape, name=name+".grad")
        PyKernels.embedding_fwd(W, idx_bt, out)
        def bwd(ctx):
            PyKernels.embedding_bwd(ctx["W"], ctx["idx"], ctx["out"])
        tape.add(name, {"W":W, "idx":idx_bt, "out":out}, bwd)
        return out

    def rmsnorm(self, tape, x, eps=1e-5, name="rmsnorm"):
        out = Tensor.zeros(x.shape, requires_grad=x.requires_grad, name=name)
        if out.requires_grad: out.grad = Tensor.zeros(out.shape, name=name+".grad")
        B,T,C = x.shape
        inv = Tensor.zeros((B*T,), requires_grad=False, name=name+".inv")
        PyKernels.rmsnorm_fwd(x, out, inv, eps=eps)
        def bwd(ctx):
            PyKernels.rmsnorm_bwd(ctx["x"], ctx["out"], ctx["inv"])
        tape.add(name, {"x":x, "out":out, "inv":inv}, bwd)
        return out

    def ffn(self, tape, x, w1, w2, name="ffn"):
        B,T,C = x.shape
        O1,C2 = w1.shape
        assert_(C==C2 and O1==4*C, "ffn w1 shape")
        O2,C3 = w2.shape
        assert_(O2==C and C3==4*C, "ffn w2 shape")
        out = Tensor.zeros((B,T,C), requires_grad=(x.requires_grad or w1.requires_grad or w2.requires_grad), name=name)
        if out.requires_grad: out.grad = Tensor.zeros(out.shape, name=name+".grad")
        act = Tensor.zeros((B,T,4*C), requires_grad=(x.requires_grad or w1.requires_grad), name=name+".act")
        act.grad = Tensor.zeros(act.shape, name=name+".act.grad")
        mask = Tensor.zeros((B,T,4*C), requires_grad=False, name=name+".mask")
        PyKernels.ffn_fwd(x, w1, w2, out, act, mask)
        def bwd(ctx):
            PyKernels.ffn_bwd(ctx["x"], ctx["w1"], ctx["w2"], ctx["out"], ctx["act"], ctx["mask"])
        tape.add(name, {"x":x, "w1":w1, "w2":w2, "out":out, "act":act, "mask":mask}, bwd)
        return out

    def attn_train(self, tape, q, k, v, n_head, head_dim, ws, name="attn_train"):
        out = Tensor.zeros(q.shape, requires_grad=(q.requires_grad or k.requires_grad or v.requires_grad), name=name)
        if out.requires_grad: out.grad = Tensor.zeros(out.shape, name=name+".grad")
        PyKernels.attn_train_fwd(q, k, v, out, ws, n_head, head_dim)
        def bwd(ctx):
            PyKernels.attn_train_bwd(ctx["q"], ctx["k"], ctx["v"], ctx["out"], ctx["ws"], n_head, head_dim)
        tape.add(name, {"q":q, "k":k, "v":v, "out":out, "ws":ws}, bwd)
        return out

    def cross_entropy(self, tape, logits, targets_bt, ws, name="cross_entropy"):
        # logits:[B,T,V] => loss:[B,T]
        B,T,V = logits.shape
        loss = Tensor.zeros((B,T), requires_grad=logits.requires_grad, name=name)
        loss.grad = Tensor.zeros((B,T), name=name+".grad")
        PyKernels.logsoftmax_nll_fwd(logits, targets_bt, loss, ws)
        def bwd(ctx):
            PyKernels.logsoftmax_nll_bwd(ctx["logits"], ctx["targets"], ctx["loss"], ctx["ws"])
        tape.add(name, {"logits":logits, "targets":targets_bt, "loss":loss, "ws":ws, "logits_ref":logits}, bwd)
        return loss

# ===================== SECTION: KV Cache (preallocated) =====================

class KVCache:
    # layout: (Tmax, n_head, B, head_dim)
    __slots__ = ("Tmax","H","B","D","K","V")
    def __init__(self, Tmax, H, B, D):
        self.Tmax, self.H, self.B, self.D = Tmax, H, B, D
        self.K = array('f', [0.0]) * (Tmax*H*B*D)
        self.V = array('f', [0.0]) * (Tmax*H*B*D)

    def _idx(self, t, h, b, i):
        # ((((t*H + h)*B + b)*D + i))
        return (((t*self.H + h)*self.B + b)*self.D + i)

    def set_kv(self, t, b, h, vec, is_k=True):
        base = self.K if is_k else self.V
        for i in range(self.D):
            base[self._idx(t,h,b,i)] = vec[i]

    def getK_accessor(self, b):
        cache = self
        class Acc:
            __slots__ = ("b",)
            def __init__(self, b): self.b = b
            def get(self, t, h, i):
                return cache.K[cache._idx(t,h,self.b,i)]
        return Acc(b)

    def getV_accessor(self, b):
        cache = self
        class Acc:
            __slots__ = ("b",)
            def __init__(self, b): self.b = b
            def get(self, t, h, i):
                return cache.V[cache._idx(t,h,self.b,i)]
        return Acc(b)

class AttnStepWorkspace:
    __slots__ = ("H","Tmax","logits","probs")
    def __init__(self, H, Tmax):
        self.H = H
        self.Tmax = Tmax
        self.logits = [array('f', [0.0]) * Tmax for _ in range(H)]
        self.probs  = [array('f', [0.0]) * Tmax for _ in range(H)]

class GPTParams:
    __slots__ = ("wte","wpe","wqkv","wo","fc1","fc2","lm_head","n_layer","n_embd")
    def __init__(self, backend, cfg: Config):
        V = cfg.vocab_size
        C = cfg.n_embd
        L = cfg.n_layer
        Tmax = cfg.block_size
        self.n_layer = L
        self.n_embd = C
        self.wte = backend.tensor_randn((V,C), std=cfg.weight_std, requires_grad=True, name="wte")
        self.wpe = backend.tensor_randn((Tmax,C), std=cfg.weight_std, requires_grad=True, name="wpe")
        # per-layer packed qkv: [3C, C]
        self.wqkv = [backend.tensor_randn((3*C,C), std=cfg.weight_std, requires_grad=True, name=f"layer{i}.wqkv") for i in range(L)]
        self.wo   = [backend.tensor_randn((C,C), std=cfg.weight_std, requires_grad=True, name=f"layer{i}.wo") for i in range(L)]
        self.fc1  = [backend.tensor_randn((4*C,C), std=cfg.weight_std, requires_grad=True, name=f"layer{i}.fc1") for i in range(L)]
        self.fc2  = [backend.tensor_randn((C,4*C), std=cfg.weight_std, requires_grad=True, name=f"layer{i}.fc2") for i in range(L)]
        self.lm_head = backend.tensor_randn((V,C), std=cfg.weight_std, requires_grad=True, name="lm_head")

    def parameters(self):
        ps = [self.wte, self.wpe, self.lm_head]
        for i in range(self.n_layer):
            ps += [self.wqkv[i], self.wo[i], self.fc1[i], self.fc2[i]]
        return ps

# ===================== SECTION: Ops (tight API, tape-recording) =====================

def split_qkv(tape, x_3c, C, name="split_qkv"):
    # x_3c: [B,T,3C] -> q,k,v: [B,T,C]
    B,T,CC = x_3c.shape
    assert_(CC == 3*C, "split_qkv: shape")
    # views by slicing last dim: contiguous here because our linear output is contiguous
    q = Tensor(x_3c.data, (B,T,C), strides=(T*CC, CC, 1), offset=x_3c.offset + 0*C,
               requires_grad=x_3c.requires_grad, base=x_3c, is_view=True, name=name+".q")
    k = Tensor(x_3c.data, (B,T,C), strides=(T*CC, CC, 1), offset=x_3c.offset + 1*C,
               requires_grad=x_3c.requires_grad, base=x_3c, is_view=True, name=name+".k")
    v = Tensor(x_3c.data, (B,T,C), strides=(T*CC, CC, 1), offset=x_3c.offset + 2*C,
               requires_grad=x_3c.requires_grad, base=x_3c, is_view=True, name=name+".v")
    # policy A: materialize to avoid strided grad mapping
    if Config.grad_view_policy == "A":
        q = q.contiguous(); k = k.contiguous(); v = v.contiguous()
    else:
        # policy B: allocate grads and map back via strided scatter-add
        for t in (q,k,v):
            if t.requires_grad:
                t.grad = Tensor.zeros(t.shape, name=t._name+".grad")
    def bwd(ctx):
        if Config.grad_view_policy == "B":
            # scatter-add q/k/v.grad into x_3c.grad
            x = ctx["x"]
            if x.grad is None:
                x.grad = Tensor.zeros(x.shape, name=x._name+".grad")
            _strided_scatter_add(x.grad, ctx["q"], 0*C)
            _strided_scatter_add(x.grad, ctx["k"], 1*C)
            _strided_scatter_add(x.grad, ctx["v"], 2*C)
    tape.add(name, {"x":x_3c, "q":q, "k":k, "v":v, "C":C}, bwd)
    return q,k,v

def _strided_scatter_add(dst_grad_3c, src_c, offsetC):
    # dst_grad_3c: [B,T,3C] contiguous, src_c: [B,T,C] contiguous
    B,T,C = src_c.shape
    CC = 3*C
    for b in range(B):
        for t in range(T):
            doff = (b*T + t)*CC + offsetC
            soff = (b*T + t)*C
            for i in range(C):
                dst_grad_3c.data[doff+i] += src_c.grad.data[soff+i]

def mean_loss(tape, loss_bt, name="mean_loss"):
    # loss_bt: [B,T] -> scalar [1]
    B,T = loss_bt.shape
    out = Tensor.zeros((1,), requires_grad=loss_bt.requires_grad, name=name)
    out.grad = Tensor.zeros((1,), name=name+".grad")
    s = 0.0
    for i in range(B*T):
        s += loss_bt.data[i]
    out.data[0] = s / (B*T)
    def bwd(ctx):
        g = ctx["out"].grad.data[0] / (B*T)
        if ctx["loss"].grad is None:
            ctx["loss"].grad = Tensor.zeros(ctx["loss"].shape, name=ctx["loss"]._name+".grad")
        for i in range(B*T):
            ctx["loss"].grad.data[i] += g
    tape.add(name, {"loss":loss_bt, "out":out}, bwd)
    return out

# ===================== SECTION: GPT Forward (train full-seq) =====================

def gpt_forward_train(backend, params: GPTParams, idx_bt, tgt_bt, cfg: Config,
                      attn_ws_by_layer, ce_ws):
    tape = Tape()
    B = len(idx_bt); T = len(idx_bt[0]); C = cfg.n_embd
    H = cfg.n_head; D = C // H

    # token + pos embeddings
    tok = backend.embedding(tape, params.wte, idx_bt, name="tok_emb")  # [B,T,C]
    # pos embedding gather: build idx for positions
    pos_idx = [list(range(T)) for _ in range(B)]
    pos = backend.embedding(tape, params.wpe, pos_idx, name="pos_emb")
    x = backend.add(tape, tok, pos, name="x_in")

    for li in range(cfg.n_layer):
        # attn block
        x_res = x
        x = backend.rmsnorm(tape, x, name=f"layer{li}.rms1")
        qkv = backend.linear(tape, x, params.wqkv[li], name=f"layer{li}.qkv")  # [B,T,3C]
        q,k,v = split_qkv(tape, qkv, C, name=f"layer{li}.splitqkv")
        attn = backend.attn_train(tape, q, k, v, H, D, attn_ws_by_layer[li], name=f"layer{li}.attn")
        attn = backend.linear(tape, attn, params.wo[li], name=f"layer{li}.wo")
        x = backend.add(tape, attn, x_res, name=f"layer{li}.res1")

        # ffn block
        x_res = x
        x = backend.rmsnorm(tape, x, name=f"layer{li}.rms2")
        x = backend.ffn(tape, x, params.fc1[li], params.fc2[li], name=f"layer{li}.ffn")
        x = backend.add(tape, x, x_res, name=f"layer{li}.res2")

    logits = backend.linear(tape, x, params.lm_head, name="lm_head")  # [B,T,V]
    loss_bt = backend.cross_entropy(tape, logits, tgt_bt, ce_ws, name="ce")
    loss = mean_loss(tape, loss_bt, name="loss_mean")
    return tape, loss

# ===================== SECTION: Optimizer (Adam, preallocated) =====================

class Adam:
    __slots__ = ("params","m","v","t","cfg")
    def __init__(self, params, cfg: Config):
        self.params = params
        self.m = [array('f', [0.0]) * p.numel() for p in params]
        self.v = [array('f', [0.0]) * p.numel() for p in params]
        self.t = 0
        self.cfg = cfg

    def step(self, lr):
        self.t += 1
        b1,b2,eps = self.cfg.beta1, self.cfg.beta2, self.cfg.eps
        for pi,p in enumerate(self.params):
            if p.grad is None: continue
            n = p.numel()
            m = self.m[pi]; v = self.v[pi]
            for i in range(n):
                g = p.grad.data[i]
                m[i] = b1*m[i] + (1-b1)*g
                v[i] = b2*v[i] + (1-b2)*g*g
                mh = m[i] / (1 - (b1 ** self.t))
                vh = v[i] / (1 - (b2 ** self.t))
                p.data[i] -= lr * mh / (math.sqrt(vh) + eps)
                p.grad.data[i] = 0.0

# ===================== SECTION: Data (tokenizer + optional multiprocessing prefetch) =====================

class CharTokenizer:
    __slots__ = ("stoi","itos","bos","vocab_size")
    def __init__(self, docs):
        uchars = sorted(set("".join(docs)))
        self.stoi = {ch:i for i,ch in enumerate(uchars)}
        self.itos = {i:ch for ch,i in self.stoi.items()}
        self.bos = len(uchars)
        self.vocab_size = len(uchars) + 1

    def encode(self, s):
        return [self.bos] + [self.stoi[ch] for ch in s] + [self.bos]

    def decode(self, ids):
        out=[]
        for i in ids:
            if i == self.bos: break
            out.append(self.itos.get(i,"?"))
        return "".join(out)

def make_batch(docs, tok: CharTokenizer, B, T):
    idx = [[tok.bos]*T for _ in range(B)]
    tgt = [[tok.bos]*T for _ in range(B)]
    for b in range(B):
        doc = random.choice(docs)
        ids = tok.encode(doc)
        n = min(T+1, len(ids))
        seq = ids[:n]
        # pad
        seq = seq + [tok.bos] * (T+1 - len(seq))
        for t in range(T):
            idx[b][t] = seq[t]
            tgt[b][t] = seq[t+1]
    return idx, tgt

class Prefetcher:
    # optional multiprocessing prefetch (default off)
    def __init__(self, docs, tok, B, T, max_q=8):
        self.q = Queue(maxsize=max_q)
        self.p = Process(target=self._worker, args=(self.q, docs, tok, B, T))
        self.p.daemon = True

    @staticmethod
    def _worker(q, docs, tok, B, T):
        random.seed(1234)
        while True:
            q.put(make_batch(docs, tok, B, T))

    def start(self):
        self.p.start()

    def get(self):
        return self.q.get()

# ===================== SECTION: Inference (step mode with KV cache) =====================

def gpt_generate_step(params: GPTParams, tok: CharTokenizer, cfg: Config, prompt_ids, max_new=32, temperature=0.8):
    # Pure python step-mode forward (no tape), with KV cache preallocated, fused step attention.
    # This is a minimal reference; uses packed weights but uses python loops.
    C = cfg.n_embd; H = cfg.n_head; D = C//H; L = cfg.n_layer
    Tmax = cfg.block_size
    B = 1
    cache = [KVCache(Tmax, H, B, D) for _ in range(L)]
    ws = [AttnStepWorkspace(H, Tmax) for _ in range(L)]

    # helpers: linear for vector
    def lin_vec(xC, W_OxC):
        O,C2 = W_OxC.shape
        y = array('f', [0.0]) * O
        for o in range(O):
            s=0.0
            woff=o*C2
            for c in range(C2):
                s += xC[c]*W_OxC.data[woff+c]
            y[o]=s
        return y

    def rms_vec(xC, eps=1e-5):
        ms=0.0
        for v in xC: ms += v*v
        ms /= len(xC)
        inv = 1.0/math.sqrt(ms+eps)
        y=array('f',[0.0])*len(xC)
        for i,v in enumerate(xC): y[i]=v*inv
        return y

    ids = list(prompt_ids)
    for t in range(min(len(ids), Tmax)):
        pass

    # run prompt to fill cache
    for pos in range(min(len(ids), Tmax)):
        token_id = ids[pos]
        # x = wte[token] + wpe[pos]
        x = array('f', [0.0]) * C
        for c in range(C):
            x[c] = params.wte.data[token_id*C+c] + params.wpe.data[pos*C+c]
        x = rms_vec(x)
        for li in range(L):
            x_res = x
            x1 = rms_vec(x)
            # qkv = x1 @ wqkv^T -> [3C]
            qkv = lin_vec(x1, params.wqkv[li])
            q = qkv[0*C:1*C]; k = qkv[1*C:2*C]; v = qkv[2*C:3*C]
            # write kv cache per head
            for h in range(H):
                cache[li].set_kv(pos, 0, h, k[h*D:(h+1)*D], is_k=True)
                cache[li].set_kv(pos, 0, h, v[h*D:(h+1)*D], is_k=False)
            # attn step
            out_heads = array('f', [0.0]) * C
            Kacc = cache[li].getK_accessor(0)
            Vacc = cache[li].getV_accessor(0)
            PyKernels.attn_step_fwd(q, Kacc, Vacc, pos, out_heads, ws[li], Tmax, D)
            # proj
            attn_out = lin_vec(out_heads, params.wo[li])
            for c in range(C):
                x[c] = attn_out[c] + x_res[c]
            # ffn
            x_res = x
            x2 = rms_vec(x)
            # fc1
            act = lin_vec(x2, params.fc1[li])  # [4C]
            for i in range(4*C):
                if act[i] < 0.0: act[i] = 0.0
            ff = lin_vec(act, params.fc2[li])  # [C]
            for c in range(C):
                x[c] = ff[c] + x_res[c]

    # sampling loop
    for pos in range(len(ids), min(len(ids)+max_new, Tmax)):
        token_id = ids[-1]
        # forward one step (similar to above but using current pos)
        x = array('f', [0.0]) * C
        for c in range(C):
            x[c] = params.wte.data[token_id*C+c] + params.wpe.data[pos*C+c]
        x = rms_vec(x)
        for li in range(L):
            x_res = x
            x1 = rms_vec(x)
            qkv = lin_vec(x1, params.wqkv[li])
            q = qkv[0*C:1*C]; k = qkv[1*C:2*C]; v = qkv[2*C:3*C]
            for h in range(H):
                cache[li].set_kv(pos, 0, h, k[h*D:(h+1)*D], is_k=True)
                cache[li].set_kv(pos, 0, h, v[h*D:(h+1)*D], is_k=False)
            out_heads = array('f', [0.0]) * C
            Kacc = cache[li].getK_accessor(0)
            Vacc = cache[li].getV_accessor(0)
            PyKernels.attn_step_fwd(q, Kacc, Vacc, pos, out_heads, ws[li], Tmax, D)
            attn_out = lin_vec(out_heads, params.wo[li])
            for c in range(C):
                x[c] = attn_out[c] + x_res[c]
            x_res = x
            x2 = rms_vec(x)
            act = lin_vec(x2, params.fc1[li])
            for i in range(4*C):
                if act[i] < 0.0: act[i] = 0.0
            ff = lin_vec(act, params.fc2[li])
            for c in range(C):
                x[c] = ff[c] + x_res[c]
        # logits
        logits = lin_vec(x, params.lm_head)  # [V]
        # sample
        V = len(logits)
        # temperature + stable softmax sampling
        m = max(logits)
        probs = [0.0]*V
        s=0.0
        for i in range(V):
            e = math.exp((logits[i]-m)/max(1e-6,temperature))
            probs[i]=e; s+=e
        r = random.random()*s
        cum=0.0
        nxt=V-1
        for i,p in enumerate(probs):
            cum += p
            if cum >= r:
                nxt=i; break
        ids.append(nxt)
        if nxt == tok.bos:
            break
    return ids

# ===================== SECTION: Main (train + inference demo) =====================

def main():
    # load tiny dataset
    if not os.path.exists('input.txt'):
        # minimal fallback without urllib (keep stdlib-only constraint: urllib is stdlib, but user requested only math/random/os/array/multiprocessing)
        # so we just create a tiny dataset if missing.
        with open('input.txt','w',encoding='utf-8') as f:
            f.write("anna\nbob\ncarla\ndavid\nemma\nfrank\n")
    docs = [line.strip() for line in open('input.txt', encoding='utf-8') if line.strip()]
    random.shuffle(docs)

    tok = CharTokenizer(docs)
    cfg = Config()
    cfg.vocab_size = tok.vocab_size

    backend = PyBackend()

    params = GPTParams(backend, cfg)
    optim = Adam(params.parameters(), cfg)

    # workspaces (by layer, by batch) to avoid per-step allocation
    B = cfg.batch_size; T = cfg.block_size
    attn_ws_by_layer = [AttnTrainWorkspace(B, cfg.n_head, T) for _ in range(cfg.n_layer)]
    ce_ws = CEWorkspace(B, T)

    # optional prefetch (off by default)
    prefetch = None  # Prefetcher(docs, tok, B, T); prefetch.start()

    for step in range(cfg.steps):
        if prefetch is None:
            idx_bt, tgt_bt = make_batch(docs, tok, B, T)
        else:
            idx_bt, tgt_bt = prefetch.get()

        tape, loss = gpt_forward_train(backend, params, idx_bt, tgt_bt, cfg, attn_ws_by_layer, ce_ws)
        tape.backward(loss)

        lr_t = cfg.lr * (1.0 - step / max(1, cfg.steps))
        optim.step(lr_t)

        if (step+1) % 10 == 0:
            print(f"step {step+1:4d}/{cfg.steps}  loss {loss.data[0]:.4f}")

    # inference
    print("\n--- inference ---")
    for _ in range(10):
        ids = gpt_generate_step(params, tok, cfg, [tok.bos], max_new=cfg.block_size, temperature=0.8)
        print(tok.decode(ids[1:]))

if __name__ == "__main__":
    main()
